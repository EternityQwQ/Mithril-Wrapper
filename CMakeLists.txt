cmake_minimum_required(VERSION 3.22.1)
project(mithril LANGUAGES CXX)

find_package(Git QUIET)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()

if(DEFINED ENV{GITHUB_SHA} AND NOT "$ENV{GITHUB_SHA}" STREQUAL "")
    string(SUBSTRING "$ENV{GITHUB_SHA}" 0 12 MITHRIL_COMMIT_ID)
elseif(GIT_FOUND)
    execute_process(
        COMMAND "${GIT_EXECUTABLE}" rev-parse --short=12 HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE MITHRIL_COMMIT_ID
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET)
    if(MITHRIL_COMMIT_ID STREQUAL "")
        set(MITHRIL_COMMIT_ID "unknown")
    endif()
else()
    set(MITHRIL_COMMIT_ID "unknown")
endif()
message(STATUS "Mithril Direct Metal build: ${MITHRIL_COMMIT_ID}")

set(CMAKE_CXX_VISIBILITY_PRESET default)
set(CMAKE_C_VISIBILITY_PRESET default)
set(CMAKE_OBJCXX_VISIBILITY_PRESET default)
set(CMAKE_VISIBILITY_INLINES_HIDDEN OFF)
if(NOT MSVC AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fvisibility=default)
endif()

option(MITHRIL_BUILD_CORE "Build the platform-neutral core" ON)
option(MITHRIL_BUILD_TESTS "Build tests" ON)
option(MITHRIL_ENABLE_SHADER_TOOLCHAIN "Build GLSL to MSL translation" ${APPLE})
option(MITHRIL_BUILD_DIRECT "Build libmithril with the Direct Metal backend" ${APPLE})

if(MITHRIL_BUILD_DIRECT AND NOT APPLE)
    message(FATAL_ERROR "MITHRIL_BUILD_DIRECT requires an Apple SDK with Metal")
endif()
if(MITHRIL_BUILD_DIRECT AND NOT MITHRIL_ENABLE_SHADER_TOOLCHAIN)
    message(FATAL_ERROR "MITHRIL_BUILD_DIRECT requires MITHRIL_ENABLE_SHADER_TOOLCHAIN")
endif()

if(MITHRIL_ENABLE_SHADER_TOOLCHAIN)
    set(ENABLE_RTTI ON CACHE BOOL "" FORCE)
    set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
    set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
    set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
    set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
    set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(Mithril-Wrapper-cpp/3rdparty/glslang)

    set(SPIRV_CROSS_STATIC ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
    set(SPIRV_HEADERS_INCLUDE_DIR
        "${CMAKE_CURRENT_SOURCE_DIR}/Mithril-Wrapper-cpp/3rdparty/SPIRV-Headers/include"
        CACHE PATH "" FORCE)
    add_subdirectory(Mithril-Wrapper-cpp/3rdparty/SPIRV-Cross)
endif()

if(MITHRIL_BUILD_CORE)
    include(cmake/direct_metal.cmake)
endif()
